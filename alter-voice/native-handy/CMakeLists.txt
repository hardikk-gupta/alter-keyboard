cmake_minimum_required(VERSION 3.22.1)
project(alter_handy C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL GGML_NATIVE GGML_OPENMP GGML_CPU_KLEIDIAI GGML_BLAS)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fvisibility=hidden")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fvisibility=hidden")
add_subdirectory(llama)
add_library(alter_handy SHARED alter_handy.cpp)
target_link_libraries(alter_handy llama android log)
target_link_options(alter_handy PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
