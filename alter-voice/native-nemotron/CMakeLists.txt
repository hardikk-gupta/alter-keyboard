cmake_minimum_required(VERSION 3.22.1)
project(alter_nemotron C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
foreach(option TRANSCRIBE_BUILD_TESTS TRANSCRIBE_BUILD_EXAMPLES TRANSCRIBE_BUILD_TOOLS TRANSCRIBE_USE_SYSTEM_BLAS TRANSCRIBE_USE_OPENMP TRANSCRIBE_BUILD_SHARED TRANSCRIBE_GGML_BACKEND_DL TRANSCRIBE_INSTALL GGML_NATIVE GGML_OPENMP GGML_CPU_KLEIDIAI)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fvisibility=hidden")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fvisibility=hidden")
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/transcribe/include")
add_subdirectory(transcribe)
add_library(alter_nemotron SHARED alter_nemotron.cpp)
target_include_directories(alter_nemotron PRIVATE transcribe/include)
target_link_libraries(alter_nemotron transcribe android log)
target_link_options(alter_nemotron PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
